***********************************数据库表结构（最新版本）****************************
1、用户表users：
CREATE TABLE users (
  id INT PRIMARY KEY AUTO_INCREMENT,
  username VARCHAR(50) UNIQUE NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  password VARCHAR(255) NOT NULL,  -- bcrypt 加密后的哈希值
  avatar VARCHAR(255) DEFAULT 'default_avatar.png',
  bio TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


2、评论表comments：
CREATE TABLE comments (
  id INT PRIMARY KEY AUTO_INCREMENT,
  content TEXT NOT NULL,
  authorId INT NOT NULL,           -- 关联 users.id
  targetId VARCHAR(100) NOT NULL DEFAULT 'default',  -- 被评论的目标ID
  image VARCHAR(255),              -- 评论主图片路径
  status ENUM('active', 'deleted') DEFAULT 'active', -- 软删除状态
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (authorId) REFERENCES users(id)
);
-- 为 authorId 和 targetId 添加索引
CREATE INDEX idx_author ON comments(authorId);
CREATE INDEX idx_target ON comments(targetId);


3、评论图片表 commentimages：
CREATE TABLE commentimages (
  id INT PRIMARY KEY AUTO_INCREMENT,
  commentId INT NOT NULL,          -- 关联 comments.id
  imageUrl VARCHAR(255) NOT NULL,  -- 图片存储路径
  filename VARCHAR(255) NOT NULL,  -- 原始文件名
  mime_type VARCHAR(100) NOT NULL DEFAULT 'image/jpeg',
  file_size INT NOT NULL DEFAULT 0,
  status ENUM('active', 'deleted') DEFAULT 'active', -- 软删除状态
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (commentId) REFERENCES comments(id) ON DELETE CASCADE
);
-- 添加索引
CREATE INDEX idx_comment ON commentimages(commentId);


4、品牌数据表（测试用）：
CREATE TABLE `品牌` (
    `品牌` VARCHAR(255) NOT NULL,
    `国家` VARCHAR(255) NOT NULL,
    `型号` VARCHAR(255) NOT NULL,
    `种类` VARCHAR(255) NOT NULL,
    `价格` INT NOT NULL,
    PRIMARY KEY (`品牌`, `型号`)
);


部分运行的小问题
    1.图片存储路径：根目录/app/upload/目录中
    2.若数据库链接失败，请检查一下端口情况。！！前端运行在5501端口，后端运行在3000端口！！
  需要修改，则要到配置文件以及前端文件中修改端口(工作任务过于庞大，建议保持此端口不要更改)
    3.在启动node.js前，请确认MySQL服务正在运行，且端口为上述要求
    4.确认一下数据库配置是否正确，不正确按照.env-->database.js-->app.js来排查，启动不成
    功也有后台显示启动失败原因以及日志
    5.建议使用微软自带的浏览器或者chorme浏览器运行网站，使用其他网站可能存在部分功能无法使
    用的情况。
